services[]:
  id           unique id (required)
  name         display name
  type         "process" (default) | "ollama"
  cwd          working folder            (process)
  command      shell command to run      (process)
  port         port the service must own; checked before start,
               passed as PORT env, used to detect "ready". If held by a
               leftover process (also the ngrokApi port), it is killed
               before start; set settings.autoKillPortConflicts=false
               to get a confirmation dialog instead
  dependsOn    ids that must be ready first; started automatically
  startAfter   ids to wait for only if they are already starting
  ngrokApi     ngrok local API url; shows the public URL
  env          extra environment variables { "KEY": "value" }
  autoStart    include in "Start all" (default true)
  model, host  Ollama model to preload and server url (ollama)

docker:
  enabled              show the Docker section
  desktopPath          Docker Desktop.exe, started when the engine is down
  autoStartContainers  container names started by "Start all"
